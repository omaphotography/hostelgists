import { supabase } from "./supabase.js" window.addEventListener("DOMContentLoaded", () => { loadHostels() }) async function loadHostels() { const container = document.getElementById("hostels") if (!container) { console.log("Container not found") return } const { data, error } = await supabase .from("hostels") .select("*") if (error) { console.log("ERROR:", error) return } console.log("HOSTELS:", data) container.innerHTML = "" for (let hostel of data) { const { data: reviews } = await supabase .from("reviews") .select("rating") .eq("hostel_id", hostel.id) let avg = 0 if (reviews && reviews.length > 0) { avg = reviews.reduce((a, b) => a + b.rating, 0) / reviews.length } container.innerHTML += `

${hostel.name}

${hostel.location}

₦${hostel.price}

${avg.toFixed(1)}

` } } window.openHostel = async (id) => { const { data: user } = await supabase.auth.getUser() if (!user.user) { window.location.href = "login.html" } else { window.location.href = `hostel.html?id=${id}` } }
